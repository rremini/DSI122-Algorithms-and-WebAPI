**Tower of Hanoi** คือ ปริศนาคณิตศาสตร์และเกมเชิงตรรกะที่ประกอบด้วยเสา 3 ต้น (เสาเริ่มต้น, เสาพัก/ตัวช่วย, และเสาปลายทาง) พร้อมด้วยแผ่นดิสก์ทรงกลมที่มีขนาดแตกต่างกันวางซ้อนกันเป็นทรงพีระมิด โดยมีเป้าหมายคือการย้ายแผ่นดิสก์ทั้งหมดจากเสาเริ่มต้นไปยังเสาปลายทาง


```JavaScript
/**
 * ฟังก์ชันแก้ปัญหา Tower of Hanoi
 * @param {number} n - จำนวนแผ่นดิสก์
 * @param {string} source - เสาเริ่มต้น
 * @param {string} target - เสาปลายทาง
 * @param {string} auxiliary - เสาพัก/ตัวช่วย
 */
function towerOfHanoi(n, source, target, auxiliary) {
    // Base Case: เมื่อเหลือแผ่นดิสก์เพียงแผ่นเดียว
    if (n === 1) {
        console.log(`ย้ายแผ่น 1 จากเสา ${source} -> ${target}`);
        return;
    }

    // ขั้นที่ 1: ย้าย n - 1 แผ่นจากเสาเริ่มต้นไปยังเสาพัก
    towerOfHanoi(n - 1, source, auxiliary, target);

    // ขั้นที่ 2: ย้ายแผ่นที่ n (แผ่นใหญ่สุดที่เหลือ) ไปยังเสาปลายทาง
    console.log(`ย้ายแผ่น ${n} จากเสา ${source} -> ${target}`);

    // ขั้นที่ 3: ย้าย n - 1 แผ่นจากเสาพักไปยังเสาปลายทาง
    towerOfHanoi(n - 1, auxiliary, target, source);
}

// ตัวอย่างการเรียกใช้งานสำหรับ 4 แผ่นดิสก์
console.log("--- ลำดับการย้ายสำหรับ 4 แผ่น ---");
towerOfHanoi(4, 'A', 'C', 'B');

// ตัวอย่างการเรียกใช้งานสำหรับ 5 แผ่นดิสก์
console.log("\n--- ลำดับการย้ายสำหรับ 5 แผ่น ---");
towerOfHanoi(5, 'A', 'C', 'B');
```
